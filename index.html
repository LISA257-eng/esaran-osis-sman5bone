const cfg = window.ESARAN_CONFIG || {}; const configured = cfg.SUPABASE_URL && cfg.SUPABASE_ANON_KEY && !cfg.SUPABASE_URL.includes('PROJECT-REF') && !cfg.SUPABASE_ANON_KEY.includes('PASTE_'); const db = configured ? window.supabase.createClient( cfg.SUPABASE_URL, cfg.SUPABASE_ANON_KEY ) : null; const form = document.querySelector('#form'); const statusBox = document.querySelector('#status'); const submit = document.querySelector('#submit'); function showStatus(message, type) { statusBox.textContent = message; statusBox.className = 'status ' + type; statusBox.style.display = 'block'; } function identity() { const selected = document.querySelector( 'input[name="identitas"]:checked' ); const open = selected && selected.value === 'terbuka'; document.querySelector('#namaField').hidden = !open; document.querySelector('#kelasField').hidden = !open; document.querySelector('#nama').required = !!open; } document.querySelectorAll('input[name="identitas"]').forEach(input => { input.addEventListener('change', identity); }); identity(); form.addEventListener('submit', async function (e) { e.preventDefault(); if (!db) { showStatus( 'Website belum terhubung ke Supabase. Periksa config.js.', 'err' ); return; } const selected = document.querySelector( 'input[name="identitas"]:checked' ); if (!selected) { showStatus('Pilih jenis identitas terlebih dahulu.', 'err'); return; } const open = selected.value === 'terbuka'; const payload = { kategori: document.querySelector('#kategori').value, anonim: !open, nama: open ? document.querySelector('#nama').value.trim() : null, kelas: open ? (document.querySelector('#kelas').value.trim() || null) : null, isi: document.querySelector('#isi').value.trim(), status: 'Baru' }; if (!payload.isi) { showStatus('Silakan isi saran terlebih dahulu.', 'err'); return; } submit.disabled = true; submit.textContent = 'Mengirim…'; showStatus('Sedang mengirim saran, tunggu sebentar…', ''); try { const { error } = await db .from('saran') .insert([payload]); if (error) { throw error; } form.reset(); identity(); showStatus( 'Terima kasih! Saranmu berhasil dikirim.', 'ok' ); } catch (err) { console.error('Gagal mengirim saran:', err); showStatus( 'Saran belum terkirim: ' + (err.message || 'Terjadi kesalahan.'), 'err' ); } finally { submit.disabled = false; submit.textContent = 'Kirim saran ↗'; } });
